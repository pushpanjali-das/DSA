struct node* create(struct node *x)
{
	if(x==NULL)
	{
		x=(struct node*)malloc(sizeof(struct node));
		printf("Enter the number\n");
		scanf("%d",&x->n);
		x->prev=NULL;
		x->next=NULL;
		return x;
	}
	else
	{
		printf("Node has been created already\n");
		return x;
	}
}

void display(struct node *x)
{
	curr=x;
	while(curr!=NULL)
	{
		printf("%d->",curr->n);
		curr=curr->next;
	}
	printf("\n");
}

void insend(struct node *x)
{
	curr=x;
	while(curr->next!=NULL)
	{
		curr=curr->next;
	}
	temp=(struct node*)malloc(sizeof(struct node));
	printf("Enter the number\n");
	scanf("%d",&temp->n);
	temp->next=NULL;
	temp->prev=curr;
	curr->next=temp;
}

struct node* insbeg(struct node *x)
{
	temp=(struct node*)malloc(sizeof(struct node));
	printf("Enter the number\n");
	scanf("%d",&temp->n);
	temp->prev=NULL;
	temp->next=x;
	x->prev=temp;
	return temp;
}

void inspos(struct node *x)
{
	int pos,c=0;
	curr=x;
	printf("Enter the position where you want to insert new node\n");
	scanf("%d",&pos);
	while(curr->next!=NULL)
	{
		c++;
		if(c==pos-1)
		{
			temp=(struct node*)malloc(sizeof(struct node));
			printf("Enter the number\n");
			scanf("%d",&temp->n);
			temp->next=curr->next;
			curr->next=temp;
			temp->prev=curr;
			break;
		}
		curr=curr->next;
	}
}
struct node* delbeg(struct node *x)
{
	curr=x;
	if(curr==NULL)
	{
		printf("No node to be deleted\n");
		return curr;
	}
	else if(curr->next==NULL)
	{
		printf("Deleting the one and only available node\n");
		free(curr);
		curr=NULL;
		return curr;
	}
	else
	{
		curr=curr->next;
		curr->prev=NULL;
		free(x);
		x=NULL;
		return curr;
	}
}

struct node* delend(struct node *x)
{
	curr=x;
	if(curr==NULL)
	{
		printf("No node to be deleted\n");
		return curr;
	}
	else if(curr->next==NULL)
	{
		printf("Deleting the one and only avaialble node\n");
		curr->next=NULL;
		free(curr);
		curr=NULL;
		return curr;
	}
	else
	{
		while(curr->next!=NULL)
		{
			pr=curr;
			curr=curr->next;
		}
		pr->next=NULL;
		free(curr);
		curr=NULL;
		curr->prev=NULL;
		return x;
	}
}

void delpos(struct node *x)
{
	int c=0,pos;
	curr=x;
	printf("Enter the position you want to delete\n");
	scanf("%d",&pos);
	while(curr->next!=NULL)
	{
		c++;
		if(c==pos-1)
		{
			curr->next->prev=NULL;
			temp=curr->next;
			curr->next=curr->next->next;
			free(temp);
			break;
		}
		curr=curr->next;
	}
}

void reverse(struct node *x)
{
	curr=x;
	while(curr->next!=NULL)
	{
		curr=curr->next;
	}

	while(curr!=NULL)
	{
		printf("%d->",curr->n);
		curr=curr->prev;
	}
	printf("\n");
}
void count(struct node *x)
{
	int c=0;
	curr=x;
	while(curr!=NULL)
	{
		c++;
		curr=curr->next;
	}
	printf("%d nodes are present in the list\n",c);
}

void search(struct node *x)
{
	int se;
	printf("Enter the searching element\n");
	scanf("%d",&se);
	curr=x;
	while(curr!=NULL)
	{
		if(se==curr->n)
		{
			printf("Element found\n");
			break;
		}
		curr=curr->next;
	}
	if(curr==NULL)
	{
		printf("Element not found\n");
	}
}

void split(struct node *x)
{
	int pos,c=0;
	curr=x;
	printf("Enter position till where you want to split\n");
	scanf("%d",&pos);
	while(curr!=NULL)
	{
		c++;
		if(c==pos)
		{
			temp=curr->next;
			curr->next=NULL;
			temp->prev=NULL;
			free(temp);
			break;
		}
		curr=curr->next;
	}
}

void merge(struct node *x, struct node *y)
{
	curr=x;
	while(curr->next!=NULL)
	{
		curr=curr->next;
	}
	curr->next=y;
	y->prev=curr;
}

void sort(struct node *x)
{
	int i,j,c=0,te;
	curr=x;
	while(curr!=NULL)
	{
		c++;
		curr=curr->next;
	}
	for(i=0;i<c-1;i++)
	{
		curr=x;
		for(j=0;j<c-i-1;j++)
		{
			if(curr->n > curr->next->n)
			{
				te=curr->n;
				curr->n=curr->next->n;
				curr->next->n=te;
			}
			curr=curr->next;
		}
	}
}
