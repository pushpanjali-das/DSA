/* QUEUE using ARRAY

-->The limitation of implementing Queue using array is, when the queue is emptied,
	the value of bot becomes greater than the top, at this point we won't be
	able to insert anymore values as the top has already reached the
	MAX_SIZE-1 location and we need excess memory to store new data even
	though the existing memory is free.
-->This Memory Wastage can be seen in Queue using array.
-->Instead we can use Circular Queue.
*/

#include<stdio.h>
#include<conio.h>
#define MAX_SIZE 5
int top=-1,bot=0;

void enqueue(int[]);
void dequeue(int[]);
void display(int[]);
void isfull();
void isempty();
int main()
{
	int a[MAX_SIZE],ch;
	clrscr();
	do
	{
		printf("1.Enqueue\n");
		printf("2.Dequeue\n");
		printf("3.Display\n");
		printf("4.IsFull\n");
		printf("5.IsEmpty\n");
		printf("6.Exit\n");
		printf("Enter your choice\n");
		scanf("%d",&ch);
		switch(ch)
		{
			case 1: enqueue(a);
					break;
			case 2: dequeue(a);
					break;
			case 3: display(a);
					break;
			case 4: isfull();
					break;
			case 5: isempty();
					break;
		}
	}while(ch!=6);
	return 0;
}

void enqueue(int x[])
{
	if(top < MAX_SIZE - 1)
	{
		top++;
		printf("Enter the number\n");
		scanf("%d",&x[top]);
	}
	else
	{
		isfull();
	}
}

void dequeue(int x[])
{
	if(top>=bot)
	{
		printf("Deleting %d\n",x[bot]);
		bot++;
	}
	else
	{
		isempty();
	}
}

void display(int x[])
{
	int i;
	for(i=bot;i<=top;i++)
	{
		printf("%d\t",x[i]);
	}
	printf("\n");
}

void isfull()
{
	if(top==MAX_SIZE-1)
	{
		printf("Queue is full\n");
	}
	else
	{
		printf("Queue is not full\n");
	}
}

void isempty()
{
	if(bot>top)
	{
		printf("Queue is empty\n");
	}
	else
	{
		printf("Queue is not empty\n");
	}
}
