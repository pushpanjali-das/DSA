/* Queue Using Double Linked List */
#include<stdio.h>
#include<conio.h>
#include<stdlib.h>

struct node *create(struct node *);
void display(struct node *);
void insend(struct node *);
struct node *delbeg(struct node *);
void isempty(struct node *);

struct node
{
	struct node *front;
	int n;
	struct node *rear;
};

struct node *curr,*temp;

int main()
{
	struct node *s;
	int ch;
	clrscr();
	s=NULL;
	do
	{
		printf("1.Create\n");
		printf("2.Display\n");
		printf("3.Insertion\n");
		printf("4.Deletion\n");
		printf("5.Isempty\n");
		printf("6.Exit\n");
		printf("Enter your choice\n");
		scanf("%d",&ch);
		switch(ch)
		{
			case 1:	s=create(s);
					break;
			case 2:	display(s);
					break;
			case 3: insend(s);
					break;
			case 4: s=delbeg(s);
					break;
			case 5: isempty(s);
					break;
		}
	}while(ch!=6);
	return 0;
}

struct node *create(struct node *x)
{
	if(x==NULL)
	{
		x=(struct node*)malloc(sizeof(struct node));
		printf("Enter the number\n");
		scanf("%d",&x->n);
		x->front=NULL;
		x->rear=NULL;
	}
	else
	{
		printf("Queue has been created already\n");
	}
	return x;
}

void display(struct node *x)
{
	curr=x;
	while(curr!=NULL)
	{
		printf("%d->",curr->n);
		curr=curr->rear;
	}
	printf("\n");
}

void insend(struct node *x)
{
	curr=x;
	while(curr->rear!=NULL)
	{
		curr=curr->rear;
	}
	temp=(struct node*)malloc(sizeof(struct node));
	printf("Enter the number\n");
	scanf("%d",&temp->n);
	curr->rear=temp;
	temp->front=curr;
	temp->rear=NULL;
}

struct node *delbeg(struct node *x)
{
	curr=x;
	if(curr==NULL)
	{
		printf("No node to be deleted\n");
		return x;
	}
	else if(curr->rear==NULL)
	{
		printf("You are deleting the one and only available node\n");
		free(curr);
		x=NULL;
		return x;
	}
	else
	{
		x=x->rear;
		free(curr);
		x->front=NULL;
		return x;
	}
}

void isempty(struct node *x)
{
	if(x==NULL)
	{
		printf("Queue is underflow (OR) Empty\n");
	}
	else
	{
		printf("Queue is not empty\n");
	}
}
