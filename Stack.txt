/* Stack is a dynamic linear data structure which is a insertion or a deletion technique */
#include<stdio.h>
#include<conio.h>
#include<stdlib.h>

struct node
{
	int n;
	struct node *next;
};

struct node *temp,*prev,*curr;
struct node* create(struct node *x);
void display(struct node *x);
void push(struct node *x);
struct node* pop(struct node *x);
void isempty(struct node *x);

int main()
{
	struct node *s;
	int ch;
	clrscr();
	s=NULL;
	do
	{
		printf("1.Create\n");
		printf("2.Display\n");
		printf("3.Push\n");
		printf("4.Pop\n");
		printf("5.Isempty\n");
		printf("6.Exit\n");
		printf("Enter the choice\n");
		scanf("%d",&ch);
		switch(ch)
		{
			case 1:
				s=create(s);
				break;
			case 2:
				display(s);
				break;
			case 3:
				push(s);
				break;
			case 4:
				s=pop(s);
				break;
			case 5:
				isempty(s);
				break;
		}
	}while(ch!=6);
	return 0;
}

struct node* create(struct node *x)
{
	if(x==NULL)
	{
		x=(struct node*)malloc(sizeof(struct node));
		printf("Enter the number\n");
		scanf("%d",&x->n);
		x->next=NULL;
		return x;
	}
	else
	{
		printf("Node has been craeted already\n");
		return x;
	}
}

void display(struct node *x)
{
	curr=x;
	while(curr!=NULL)
	{
		printf("%d->",curr->n);
		curr=curr->next;
	}
	printf("\n");
}

void push(struct node *x)
{
	curr=x;
	while(curr->next != NULL)
	{
		curr=curr->next;
	}
	temp=(struct node*)malloc(sizeof(struct node));
	printf("Enter the number to be pushed\n");
	scanf("%d",&temp->n);
	temp->next=NULL;
	curr->next=temp;
}

struct node* pop(struct node *x)
{
	curr=x;
	if(curr==NULL)
	{
		printf("No node to be deleted\n");
		return x;
	}
	else if(curr->next==NULL)
	{
		printf("Deleting the one and only available node\n");
		free(curr);
		x=NULL;
		return x;
	}
	else
	{
		while(curr->next!=NULL)
		{
			prev=curr;
			curr=curr->next;
		}
		prev->next=NULL;
		free(curr);
		curr=NULL;
		return x;
	}
}
void isempty(struct node *x)
{
	if(x==NULL)
	{
		printf("Stack is empty\n");
	}
	else
	{
		printf("Stack is not empty\n");
	}
}
