struct node *create(struct node *x)
{
	if(x==NULL)
	{
		x=(struct node*)malloc(sizeof(struct node));
		printf("Enter the number\n");
		scanf("%d",&x->n);
		x->next=x;
	}
	else
	{
		printf("List has been created already\n");
	}
	return x;
}

void display(struct node *x)
{
	curr=x;
	while(curr->next!=x)
	{
		printf("%d->",curr->n);
		curr=curr->next;
	}
	printf("%d->\n",curr->n);
}

void insend(struct node *x)
{
	curr=x;
	while(curr->next != x)
	{
		curr=curr->next;
	}
	temp=(struct node*)malloc(sizeof(struct node));
	printf("Enter the number\n");
	scanf("%d",&temp->n);
	curr->next=temp;
	temp->next=x;
}

void inspos(struct node *x)
{
	int pos,c=0;
	curr=x;
	printf("Enter the position to be inserted\n");
	scanf("%d",&pos);
	while(curr->next!=x)
	{
		c++;
		if(c==pos-1)
		{
			temp=(struct node*)malloc(sizeof(struct node));
			printf("Enter the number\n");
			scanf("%d",&temp->n);
			temp->next=curr->next;
			curr->next=temp;
			break;
		}
		curr=curr->next;
	}
}

struct node *delend(struct node *x)
{
	struct node *pr;
	curr=x;
	if(curr==NULL)
	{
		printf("No node to be deleted\n");
	}
	else if(curr->next == x)
	{
		printf("You deleting the one and only available node\n");
		free(curr);
		x=NULL;
	}
	else
	{
		while(curr->next != x)
		{
			pr=curr;
			curr=curr->next;
		}
		free(curr);
		pr->next=x;
	}
	return x;
}

struct node* merge(struct node *x, struct node *y)
{
	struct node *pr;
	curr=x;
	while(curr->next != x)
	{
		curr=curr->next;
	}
	curr->next=y;
	curr=curr->next;
	while(curr->next != y)
	{
		curr=curr->next;
	}

}

void sort(struct node *x)
{
	int i=0,j=0,c=0,te;
	curr=x;
	while(curr->next != x)
	{
		c++;
		curr=curr->next;
	}
	c++;
	for(i=0;i<c-1;i++)
	{
		curr=x;
		for(j=0;j<c-i-1;j++)
		{
			if(curr->n > curr->next->n)
			{
				te=curr->n;
				curr->n=curr->next->n;
				curr->next->n=te;
			}
			curr=curr->next;
		}
	}
}

void reverse(struct node *x)
{
	struct node *last;
	curr=x;
	while(curr->next != x)
	{
		curr=curr->next;
	}
	printf("%d->",curr->n);
	last=curr;
	do
	{
		curr=x;
		while(curr->next != last)
		{
			curr=curr->next;
		}
		last=curr;
		printf("%d->",curr->n);
	}while(last != x);
