/* Linear Search Model-1:
   Searching for the element until the element is found (or) till the end of the array.
 */

#include<stdio.h>
#include<conio.h>

int main()
{
	int a[10],n,i,se;
	clrscr();
	printf("Enter the number of variables to be used\n");
	scanf("%d",&n);
	for(i=0;i<n;i++)
	{
		printf("Enter the value of a[%d]\n",i);
		scanf("%d",&a[i]);
	}
	printf("Enter the element to be searched\n");
	scanf("%d",&se);
	for(i=0;i<n;i++)
	{
		if(se==a[i])
		{
			printf("Element has been found at index %d\n",i);
			break;
		}
	}
	if(i==n)
	{
		printf("Element not found\n");
	}
	return 0;
}
