/* Bubble sort */
#include<stdio.h>
#include<conio.h>

int main()
{
	int a[10],n,i,j,temp;
	clrscr();
	printf("Enter the number of variables to be used\n");
	scanf("%d",&n);
	for(i=0;i<n;i++)
	{
		printf("Enter the value of a[%d]\n",i);
		scanf("%d",&a[i]);
	}
	for(i=0;i<n-1;i++)  // Total n-1 passes will be there
	{
		for(j=0;j<n-i-1;j++) // In each pass there will total n-i-1 comparisons
		{
			if(a[j]>a[j+1])
			{
				temp=a[j];
				a[j]=a[j+1];
				a[j+1]=temp;
			}
		}
	}
	printf("Sorted array is\n");
	for(i=0;i<n;i++)
	{
		printf("a[%d]=%d\n",i,a[i]);
	}
	return 0;
}
